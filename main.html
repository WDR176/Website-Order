var tdsEntries = entries.Where(e => string.Equals(e.TaxType, "TDS", StringComparison.OrdinalIgnoreCase)).ToList(); var tcsEntries = entries.Where(e => string.Equals(e.TaxType, "TCS", StringComparison.OrdinalIgnoreCase)).ToList(); var responseData = new LedgerImportData { EntityId = entityId, FileName = fileName, LedgerType = ledgerType, ImportedAt = DateTime.UtcNow.ToString("o"), ProcessingTimeSeconds = Math.Round(processingTime, 2), EntryCount = entries.Count, // TDS totals TdsCount = tdsEntries.Count, TotalTdsAmount = tdsEntries.Sum(e => e.TdsAmount), TotalCreditAmount = tdsEntries.Sum(e => e.CreditAmount), // TCS totals TcsCount = tcsEntries.Count, TotalTcsAmount = tcsEntries.Sum(e => e.TcsAmount), TotalSalesAmount = tcsEntries.Sum(e => e.SalesAmount), LedgerEntries = entries }; _logger.LogInformation( $"Import done [{ledgerType}]: {entries.Count} entries | " + $"TDS={responseData.TotalTdsAmount:N2} ({tdsEntries.Count}) | " + $"TCS={responseData.TotalTcsAmount:N2} ({tcsEntries.Count}) | " + $"{processingTime:F2}s"); return new OkObjectResult( LedgerImportResponse.Success(responseData, $"{ledgerType} ledger imported — {entries.Count} entries.")); } catch (Exception ex) { _logger.LogError(ex, "Unexpected error in ImportLedgerFromExcel"); return new ObjectResult(LedgerImportResponse.Failure(ex)) { StatusCode = 500 }; } } // ── Helper ──────────────────────────────────────────────────────────── private static BadRequestObjectResult Fail(string code, string message) => new(LedgerImportResponse.Failure(code, message));
